cmake_minimum_required(VERSION 3.22)
project(RIDENCore VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(RIDEN_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(riden_core SHARED src/platform.c src/rtu.c src/core.c)
target_include_directories(riden_core PUBLIC include PRIVATE src)
target_compile_definitions(riden_core PRIVATE RIDEN_BUILD _POSIX_C_SOURCE=200809L)
if(MSVC)
  target_compile_options(riden_core PRIVATE /W4)
else()
  target_compile_options(riden_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(riden_core PRIVATE Threads::Threads)
endif()
if(RIDEN_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(riden_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(riden_core PRIVATE -fsanitize=address,undefined)
endif()
if(ANDROID)
  target_sources(riden_core PRIVATE src/android_jni.c)
else()
  if(WIN32)
    target_sources(riden_core PRIVATE src/serial_win32.c)
  else()
    target_sources(riden_core PRIVATE src/serial_posix.c)
  endif()
  add_executable(riden-cored src/daemon.c src/ipc.c)
  target_include_directories(riden-cored PRIVATE include src)
  target_link_libraries(riden-cored PRIVATE riden_core)
  if(WIN32)
    target_link_libraries(riden-cored PRIVATE ws2_32)
  endif()
  if(UNIX)
    set_target_properties(riden-cored PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  add_executable(riden-bench src/bench.c)
  target_include_directories(riden-bench PRIVATE include src)
  target_link_libraries(riden-bench PRIVATE riden_core)
  if(UNIX)
    set_target_properties(riden-bench PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID AND NOT CMAKE_CROSSCOMPILING)
  add_executable(test-core tests/test_core.c)
  target_include_directories(test-core PRIVATE include src)
  target_link_libraries(test-core PRIVATE riden_core)
  add_test(NAME core COMMAND test-core)
  if(UNIX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME daemon-pty COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:riden-cored>)
  endif()
endif()
if(NOT ANDROID)
  install(TARGETS riden_core riden-cored riden-bench RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
  install(FILES include/riden_core.h DESTINATION include)
endif()
